E           def compute_sha256(path: Path) -> str:
E               h = hashlib.sha256()
E               with path.open("rb") as f:
E                   while chunk := f.read(65536):
E                       h.update(chunk)
E               return h.hexdigest()

E                           if node.type is None:

E                           if len(node.body) == 1 and isinstance(node.body[0], ast.Pass):
E                               violations.append(f"{pf.name}:{node.lineno}: except ...: pass")
E           
E                       if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
E                           for arg in node.args.args + node.args.kwonlyargs:
E                               if "allow_unauth" in arg.arg.lower():
E                                   violations.append(f"{pf.name}:{node.lineno}: parameter {arg.arg}")
E           
E                       if isinstance(node, ast.Name) and node.id == "DEFAULT_SITE_SALT":
E                           violations.append(f"{pf.name}:{node.lineno}: DEFAULT_SITE_SALT found")
E           
E                       if isinstance(node, ast.Import):
E                           for alias in node.names:
E                               if alias.name == "pickle":
E                                   violations.append(f"{pf.name}:{node.lineno}: pickle import")
E                       if isinstance(node, ast.ImportFrom) and node.module == "pickle":
E                           violations.append(f"{pf.name}:{node.lineno}: pickle import")
E           
E                       if isinstance(node, ast.Call) and isinstance(node.func, ast.Name):
E                           if node.func.id in ("eval", "exec"):
E                               violations.append(f"{pf.name}:{node.lineno}: call to {node.func.id}()")
E           
E               details = {"violations": violations, "scanned_files": len(py_files)}
E               passed = len(violations) == 0
E               status = "PASS" if passed else "FAIL"
